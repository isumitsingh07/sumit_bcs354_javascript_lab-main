    function runCodeLive() {
        const assign = LAB_DATA[currentAssignIdx];
        const task = assign.tasks[currentTaskIdx];
        const termOutput = document.getElementById('termOutput');

        if (task.isHtml) {
            document.getElementById('previewFrame').src = task.htmlPath;
            return;
        }

        let logs = [];
        const originalLog = console.log;
        const originalError = console.error;

        console.log = (...args) => {
            logs.push(args.map(a => typeof a === 'object' ? JSON.stringify(a) : a).join(' '));
        };
        console.error = (...args) => {
            logs.push("ERROR: " + args.join(' '));
        };

        try {
            const executable = task.code.replace(/\/\*[\s\S]*?\*\//g, '');
            const runFn = new Function(executable);
            runFn();

            setTimeout(() => {
                console.log = originalLog;
                console.error = originalError;
                if (logs.length > 0) {
                    termOutput.textContent = logs.join('\n');
                } else {
                    termOutput.textContent = task.output;
                }
            }, 150);
        } catch (e) {
            console.log = originalLog;
            console.error = originalError;
            termOutput.textContent = "Output:\n" + task.output;
        }
    }
